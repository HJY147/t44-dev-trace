import ast
import hashlib
import json
import subprocess
import xml.etree.ElementTree as ET
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path.cwd()
E = '6fa21f473f78484fa3a7103abf42eab166bbb4c9'
SOURCE = '3fc58d18614c394cc588a16b316cf89c1810ecf1'
OLD = 'c4e8c696411975ee6964ef3b2254c8f3937ecb32'
C = '0dfb55c2191200d947c5423fd9ee74bb0812cc5a'
BASE = 'agents/database/output-guard/'
TOOL = 'tmp/838_database_public_diagnostics/838_database_public_diagnostics.py'
TEST = 'tmp/836_database_output_guard/836_output_guard.py'

def blob(commit, name):
    return subprocess.check_output(['git', 'show', f'{commit}:{name}'], cwd=ROOT)

def digest(data):
    return hashlib.sha256(data).hexdigest()

def require(condition, name):
    if not condition:
        raise ValueError(name)

def function_source(data, name):
    text = data.decode()
    nodes = [n for n in ast.parse(text).body if isinstance(n, ast.FunctionDef) and n.name == name]
    require(len(nodes) == 1, name)
    return ast.get_source_segment(text, nodes[0])

seal = json.loads(blob(E, BASE + 'evidence-seal.json'))
file_checks = []
for name, expected in seal['files_sha256'].items():
    actual = digest(blob(E, BASE + name))
    require(actual == expected, name)
    file_checks.append({'path': BASE + name, 'sha256': actual, 'matches': True})
source_checks = []
for name, expected in seal['source_files_sha256'].items():
    fixed, final = blob(SOURCE, name), blob(E, name)
    require(fixed == final and digest(final) == expected, name)
    source_checks.append({'path': name, 'sha256': digest(final), 'SOURCE_E_identical': True})
verification = json.loads(blob(E, BASE + 'verification.json'))
old_checks = []
for item in verification['original_c4e8_evidence']:
    name = item['path']
    old, final = blob(OLD, name), blob(E, name)
    require(old == final and digest(final) == item['sha256'], name)
    old_checks.append({'path': name, 'sha256': digest(final), 'OLD_E_identical': True})

tool_old, tool_new = blob(OLD, TOOL), blob(SOURCE, TOOL)
functions_unchanged = {}
for name in ('recipes', 'provenance'):
    functions_unchanged[name] = function_source(tool_old, name) == function_source(tool_new, name)
    require(functions_unchanged[name], name)
tree = ast.parse(tool_new)
constants = {n.targets[0].id: ast.literal_eval(n.value) for n in tree.body
             if isinstance(n, ast.Assign) and isinstance(n.targets[0], ast.Name)
             and n.targets[0].id in ('SOURCE', 'FILES')}
require(constants['SOURCE'] == C and len(constants['FILES']) == 13, 'source binding')
bound_files = []
for name in constants['FILES']:
    path = 'backend/t44/' + name
    original, final = blob(C, path), blob(E, path)
    require(original == final, path)
    bound_files.append({'path': path, 'sha256': digest(final), 'C_E_identical': True})
production_diff = subprocess.check_output(['git', 'diff', '--name-only', OLD, E, '--',
    'backend/t44', 'frontend/src', 'specs', 'database'], cwd=ROOT, text=True).splitlines()
require(not production_diff, 'production delta')
source_checkpoint_paths = subprocess.check_output(['git', 'diff-tree', '--no-commit-id',
    '--name-only', '-r', SOURCE], cwd=ROOT, text=True).splitlines()
require(set(source_checkpoint_paths) == {TOOL, TEST}, 'source checkpoint changed files')

def junit(name, expected):
    data = blob(E, BASE + name)
    root = ET.fromstring(data)
    suites = [s for s in root.iter('testsuite') if not s.findall('testsuite')]
    totals = {key: sum(int(s.attrib.get(key, '0')) for s in suites)
              for key in ('tests', 'failures', 'errors', 'skipped')}
    require(totals == expected, name + ' counts')
    cases = []
    for case in root.iter('testcase'):
        cases.append({'name': case.attrib['name'],
                      'failure': case.find('failure') is not None,
                      'error': case.find('error') is not None,
                      'properties': {p.attrib['name']: p.attrib.get('value', '')
                                     for p in case.findall('./properties/property')}})
    require(len(cases) == expected['tests'], name + ' cases')
    return {'path': BASE + name, 'sha256': digest(data), 'totals': totals, 'cases': cases}

junit_checks = [
    junit('red-junit.xml', {'tests': 2, 'failures': 0, 'errors': 2, 'skipped': 0}),
    junit('red-run2-junit.xml', {'tests': 2, 'failures': 2, 'errors': 0, 'skipped': 0}),
    junit('green-junit.xml', {'tests': 11, 'failures': 0, 'errors': 0, 'skipped': 0}),
]
for index, expected in ((1, {'main_exit': '0', 'overwritten': 'True'}),
                        (2, {'main_exit': '2', 'overwritten': 'False'})):
    case = next(c for c in junit_checks[index]['cases']
                if c['name'] == 'test_result_appears_after_old_precheck')
    require(case['properties'] == expected, 'counterexample property')
diff_receipt = json.loads(blob(E, BASE + 'diff-check-result.json'))

result = {
    'verified_at_utc': datetime.now(timezone.utc).isoformat(),
    'review_target_E': E, 'source_checkpoint': SOURCE, 'source_C': C,
    'mechanical_only': True, 'reviewer_test_execution_count': 0,
    'reviewer_matrix_DB_HTTP_load_CI_runs': 0,
    'manifest_file_checks': file_checks, 'source_checks': source_checks,
    'old_evidence_preservation_checks': old_checks,
    'recipes_provenance_literal_source_unchanged': functions_unchanged,
    'C_bound_source_files': bound_files, 'production_delta_paths_OLD_E': production_diff,
    'source_checkpoint_paths': source_checkpoint_paths, 'owner_junit_recalculated': junit_checks,
    'owner_diff_check_receipt': diff_receipt,
    'independent_private_material_read': False,
    'note': 'Owner executions are recalculated from public JUnit; the reviewer did not execute target, tests, recipes, DB or network.'
}
out = ROOT / 'agents/reviewer/evidence/database-output-guard/verification.json'
out.write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n')
print(json.dumps({'status': 'verified', 'manifest_files': len(file_checks),
    'source_files': len(source_checks), 'old_evidence': len(old_checks),
    'C_bound': len(bound_files), 'owner_junit': [j['totals'] for j in junit_checks],
    'reviewer_test_execution_count': 0}, ensure_ascii=False))
