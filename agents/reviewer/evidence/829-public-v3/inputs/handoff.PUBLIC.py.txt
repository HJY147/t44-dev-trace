"""PUBLIC neutral project-scope provisioning handshake. No acceptance cases.
Only independent operator creates fixtures via authority APIs after admission.
Root provisions exact requested existing IDs and seals credentials as opaque bytes.
RSA-OAEP(SHA256) wraps a random AES256/HMAC key, no secret argv/env/files.
"""
from __future__ import annotations
import argparse,base64,ctypes,hashlib,hmac,json,os,secrets,subprocess,sys,tempfile,time
from pathlib import Path
from control import BoundaryError,aes_cbc,canonical,utc

def atomic_json(path,value):
    temp=Path(str(path)+'.pending');temp.write_text(json.dumps(value,indent=2));temp.replace(path)

def _rsa(data,keypath,decrypt=False):
    # Convert the existing PEM key to Apple's documented RSA DER representation.
    # The system Security framework performs RSA-OAEP with SHA256 (LibreSSL CLI
    # in this environment does not support rsa_oaep_md:sha256).
    command=['/usr/bin/openssl','rsa','-in',str(keypath),'-outform','DER']
    if not decrypt:command+=['-pubin','-RSAPublicKey_out']
    r=subprocess.run(command,capture_output=True,timeout=10)
    if r.returncode:raise BoundaryError('RSA key conversion failed')
    cf=ctypes.CDLL('/System/Library/Frameworks/CoreFoundation.framework/CoreFoundation');sec=ctypes.CDLL('/System/Library/Frameworks/Security.framework/Security');vp=ctypes.c_void_p
    cf.CFDataCreate.argtypes=[vp,vp,ctypes.c_long];cf.CFDataCreate.restype=vp
    cf.CFDictionaryCreate.argtypes=[vp,ctypes.POINTER(vp),ctypes.POINTER(vp),ctypes.c_long,vp,vp];cf.CFDictionaryCreate.restype=vp
    cf.CFDataGetLength.argtypes=[vp];cf.CFDataGetLength.restype=ctypes.c_long;cf.CFDataGetBytePtr.argtypes=[vp];cf.CFDataGetBytePtr.restype=vp;cf.CFRelease.argtypes=[vp]
    sec.SecKeyCreateWithData.argtypes=[vp,vp,ctypes.POINTER(vp)];sec.SecKeyCreateWithData.restype=vp
    operation=sec.SecKeyCreateDecryptedData if decrypt else sec.SecKeyCreateEncryptedData;operation.argtypes=[vp,vp,vp,ctypes.POINTER(vp)];operation.restype=vp
    const=lambda name:vp.in_dll(sec,name).value
    keys=(vp*2)(const('kSecAttrKeyType'),const('kSecAttrKeyClass'));values=(vp*2)(const('kSecAttrKeyTypeRSA'),const('kSecAttrKeyClassPrivate' if decrypt else 'kSecAttrKeyClassPublic'))
    objects=[];error=vp()
    try:
        attributes=cf.CFDictionaryCreate(None,keys,values,2,None,None);objects.append(attributes);keydata=cf.CFDataCreate(None,r.stdout,len(r.stdout));objects.append(keydata);key=sec.SecKeyCreateWithData(keydata,attributes,ctypes.byref(error))
        if not key:raise BoundaryError('native RSA import failed')
        objects.append(key);payload=cf.CFDataCreate(None,data,len(data));objects.append(payload);result=operation(key,const('kSecKeyAlgorithmRSAEncryptionOAEPSHA256'),payload,ctypes.byref(error))
        if not result:raise BoundaryError('native RSA-OAEP-SHA256 failed')
        objects.append(result);return ctypes.string_at(cf.CFDataGetBytePtr(result),cf.CFDataGetLength(result))
    finally:
        if error.value:cf.CFRelease(error)
        for obj in reversed(objects):
            if obj:cf.CFRelease(obj)

def seal(request,credential):
    if request.get('kind')!='SERVICE_SCOPE_REQUEST' or not request.get('nonce') or not request.get('project_ids'):raise BoundaryError('scope request missing actual IDs/nonce')
    if set(credential.get('project_ids',[]))!=set(request['project_ids']) or credential.get('scope')!=['escalation:write'] or len(credential.get('token',''))<32:raise BoundaryError('credential scope/token mismatch')
    key=secrets.token_bytes(64);iv=secrets.token_bytes(16);plain=canonical({'nonce':request['nonce'],'project_ids':request['project_ids'],'credential':credential});cipher=aes_cbc(plain,key[:32],iv);binding=canonical({'nonce':request['nonce'],'project_ids':request['project_ids'],'context':request.get('context',{})});tag=hmac.new(key[32:],binding+iv+cipher,hashlib.sha256).hexdigest()
    with tempfile.TemporaryDirectory() as temp:
        pub=Path(temp)/'public.pem';pub.write_text(request['public_key_pem']);wrapped=_rsa(key,pub)
    return {'kind':'SEALED_SERVICE_SCOPE','nonce':request['nonce'],'project_ids':request['project_ids'],'context':request.get('context',{}),'wrapped_key_b64':base64.b64encode(wrapped).decode(),'iv_b64':base64.b64encode(iv).decode(),'ciphertext_b64':base64.b64encode(cipher).decode(),'hmac_sha256':tag,'sealed_utc':utc()}

class FixtureHandoff:
    def __init__(self,workspace,deadline,journal,binding=None):
        self.root=Path(workspace);self.deadline=deadline;self.journal=journal;self.binding=binding or {};self.public=self.root/'PUBLIC/handoff';self.public.mkdir(exist_ok=True);self.private=self.root/'private/secrets';self.private.mkdir(exist_ok=True,mode=0o700);self.credentials={}
        self.keypath=self.private/('handoff-'+secrets.token_hex(8)+'.pem')
        r=subprocess.run(['/usr/bin/openssl','genpkey','-algorithm','RSA','-pkeyopt','rsa_keygen_bits:3072','-out',str(self.keypath)],capture_output=True,timeout=20)
        if r.returncode:raise BoundaryError('handoff key generation failed')
        os.chmod(self.keypath,0o600);r=subprocess.run(['/usr/bin/openssl','pkey','-in',str(self.keypath),'-pubout'],capture_output=True,text=True,timeout=10)
        if r.returncode:raise BoundaryError('handoff public key failed')
        self.public_key=r.stdout
    def request(self,project_id,iteration_id=None):
        if type(project_id) is not int or project_id<1:raise BoundaryError('actual created project ID required')
        if project_id in self.credentials:return self.credentials[project_id]
        nonce=secrets.token_hex(24);request={'kind':'SERVICE_SCOPE_REQUEST','context':self.binding,'nonce':nonce,'project_ids':[project_id],'iteration_ids':[] if iteration_id is None else [iteration_id],'scope':['escalation:write'],'requested_utc':utc(),'public_key_pem':self.public_key,'purpose':'neutral service/scheduler scope binding only; no case content','root_action':'provision exact actual project IDs; apply to scheduler identity; return sealed credential'}
        path=self.public/(nonce+'.request.PUBLIC.json');atomic_json(path,request);self.journal.append({'kind':'fixture_scope_request',**request});reply=self.public/(nonce+'.sealed.PUBLIC.json');deadline=min(self.deadline,time.monotonic()+120)
        while time.monotonic()<deadline:
            if reply.exists():
                envelope=json.loads(reply.read_text())
                if envelope.get('nonce')!=nonce or envelope.get('project_ids')!=[project_id] or envelope.get('context',{})!=self.binding:raise BoundaryError('sealed scope response binding mismatch')
                key=_rsa(base64.b64decode(envelope['wrapped_key_b64']),self.keypath,True);iv=base64.b64decode(envelope['iv_b64']);cipher=base64.b64decode(envelope['ciphertext_b64']);binding=canonical({'nonce':nonce,'project_ids':[project_id],'context':self.binding});tag=hmac.new(key[32:],binding+iv+cipher,hashlib.sha256).hexdigest()
                if not hmac.compare_digest(tag,envelope['hmac_sha256']):raise BoundaryError('sealed scope response authentication failed')
                value=json.loads(aes_cbc(cipher,key[:32],iv,True));credential=value['credential']
                if value.get('nonce')!=nonce or value.get('project_ids')!=[project_id] or credential.get('project_ids')!=[project_id] or credential.get('scope')!=['escalation:write'] or len(credential.get('token',''))<32 or 'scheduler' not in credential.get('applied_to',[]):raise BoundaryError('exact service/scheduler credential metadata required')
                self.credentials[project_id]=credential;self.journal.append({'kind':'fixture_scope_received','project_ids':[project_id],'credential':credential,'utc':utc()});return credential
            time.sleep(min(.1,max(0,deadline-time.monotonic())))
        raise BoundaryError('service scope handoff absent/expired; no guessed IDs/token')

def main():
    p=argparse.ArgumentParser();p.add_argument('--request',required=True);p.add_argument('--output',required=True);a=p.parse_args();request=json.loads(Path(a.request).read_text());credential=json.load(sys.stdin);out=seal(request,credential);atomic_json(Path(a.output),out);print(json.dumps({'sealed_output':a.output,'nonce':request['nonce'],'project_ids':request['project_ids']}))
if __name__=='__main__':main()
