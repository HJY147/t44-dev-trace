def format_utc(value):return value.astimezone(dt.timezone.utc).isoformat(timespec='microseconds').replace('+00:00','Z')
def paired_clock_sample(clock):
    # before/UTC/after are one ordered sample. Use BEFORE when converting a
    # UTC expiry into a conservative monotonic deadline, never a later anchor.
    before=clock.monotonic();now=clock.utc_now();after=clock.monotonic()
    if now.tzinfo is None or after<before:raise BoundaryError('invalid clock sample')
    return {'mono_before':before,'utc':format_utc(now),'mono_after':after,'sample_width_seconds':after-before}
def conservative_deadline_anchor(clock,expires,wall_seconds,outer_deadline=None):
    sample=paired_clock_sample(clock);remaining=(expires-parse_utc(sample['utc'])).total_seconds()
    deadline=min(sample['mono_before']+float(wall_seconds),sample['mono_before']+max(0,remaining))
    if outer_deadline is not None:deadline=min(deadline,float(outer_deadline))
    return {**sample,'expires_at':format_utc(expires),'deadline_monotonic':deadline,'conversion_anchor':'mono_before','expired':sample['mono_after']>=deadline or remaining<=0}
def dual_deadline_sample(clock,expires,deadline):
    sample=paired_clock_sample(clock);now=parse_utc(sample['utc']);utc_late=(now-expires).total_seconds();mono_late=sample['mono_after']-deadline
    return {**sample,'expires_at':format_utc(expires),'deadline_monotonic':deadline,'utc_expired':utc_late>=0,'mono_expired':mono_late>=0,'expired':utc_late>=0 or mono_late>=0,'utc_overshoot_seconds':max(0,utc_late),'mono_overshoot_seconds':max(0,mono_late)}

def aes_cbc(data,key,iv,decrypt=False):
    if len(key)!=32 or len(iv)!=16:raise BoundaryError('invalid AES key/IV size')
    out=ctypes.create_string_buffer(len(data)+32);written=ctypes.c_size_t()
    rc=_crypto.CCCrypt(int(decrypt),0,1,key,len(key),iv,data,len(data),out,len(out),ctypes.byref(written))
    if rc:raise BoundaryError('CommonCrypto AES error '+str(rc))
    return out.raw[:written.value]

class EncryptedJournal:
    """Durable frames: streaming zlib -> AES256-CBC(random IV)/HMAC-SHA256.
    HMAC authenticates header/index/previous tag/ciphertext before decryption.
    Each append fsyncs; crash recovery authenticates a prefix, never claims complete.
    No entire ciphertext is loaded when closing, hashing or recovering.
    """
    def __init__(self,path:Path,secret_dir:Path,compressed_limit=MAX_EVIDENCE_COMPRESSED_BYTES,plaintext_limit=MAX_EVIDENCE_PLAINTEXT_BYTES,ciphertext_limit=MAX_EVIDENCE_CIPHERTEXT_BYTES):
        path.parent.mkdir(parents=True,exist_ok=True,mode=0o700);secret_dir.mkdir(parents=True,exist_ok=True,mode=0o700)
        self.path=path;self.lock=threading.Lock();self.n=0;self.plaintext_bytes=0;self.compressed_bytes=0;self.frames=0;self.previous=bytes(32);self.closed=False;self.ciphertext_bytes=0;self.ciphertext_limit=min(int(ciphertext_limit),MAX_EVIDENCE_CIPHERTEXT_BYTES)
        self.compressed_limit=min(int(compressed_limit),MAX_EVIDENCE_COMPRESSED_BYTES);self.plaintext_limit=min(int(plaintext_limit),MAX_EVIDENCE_PLAINTEXT_BYTES)
        if min(self.compressed_limit,self.plaintext_limit,self.ciphertext_limit)<=0:raise BoundaryError('nonpositive evidence budget')
        self.key=secrets.token_bytes(64);self.keypath=secret_dir/(path.name+'.'+secrets.token_hex(16)+'.key')
        fd=os.open(self.keypath,os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o600)
        with os.fdopen(fd,'wb') as f:f.write(self.key);f.flush();os.fsync(f.fileno())
        self.fd=os.open(path,os.O_WRONLY|os.O_CREAT|os.O_EXCL,0o600);self.sha=hashlib.sha256();self.filemac=hmac.new(self.key[32:],digestmod=hashlib.sha256)
        self.z=zlib.compressobj(6,zlib.DEFLATED,zlib.MAX_WBITS);self._write(MAGIC)
    def _write(self,data):
        if self.ciphertext_bytes+len(data)>self.ciphertext_limit:raise BoundaryError('ciphertext disk budget exhausted; diagnostic/incomplete only')
        view=memoryview(data)
        while view:
            n=os.write(self.fd,view)
            if n<=0:raise BoundaryError('evidence write failed')
            self.sha.update(view[:n]);self.filemac.update(view[:n]);self.ciphertext_bytes+=n;view=view[n:]
    def _frame(self,compressed):
        if not compressed:return
        if self.compressed_bytes+len(compressed)>self.compressed_limit:raise BoundaryError('compressed evidence budget exhausted; diagnostic/incomplete only')
        self.frames+=1;iv=secrets.token_bytes(16);cipher=aes_cbc(compressed,self.key[:32],iv);header=FRAME.pack(self.frames,len(cipher),iv)
        tag=hmac.new(self.key[32:],MAGIC+header+self.previous+cipher,hashlib.sha256).digest()
        if self.ciphertext_bytes+len(header)+len(cipher)+len(tag)>self.ciphertext_limit:raise BoundaryError('ciphertext disk frame reservation failed; incomplete only')
        self._write(header);self._write(cipher);self._write(tag);self.previous=tag;self.compressed_bytes+=len(compressed)
    def append(self,value):
        line=canonical(value)+b'\n'
        with self.lock:
            if self.closed:raise BoundaryError('journal closed')
            if self.plaintext_bytes+len(line)>self.plaintext_limit:raise BoundaryError('plaintext evidence safety limit; diagnostic/incomplete only')
            for offset in range(0,len(line),65536):self._frame(self.z.compress(line[offset:offset+65536]))
            self._frame(self.z.flush(zlib.Z_SYNC_FLUSH));os.fsync(self.fd)
            self.n+=1;self.plaintext_bytes+=len(line)
    def close(self):
        with self.lock:
            if self.closed:return self.integrity
            self._frame(self.z.flush(zlib.Z_FINISH));os.fsync(self.fd);os.close(self.fd);self.closed=True
            result={'format':'I829J3','cipher':'AES-256-CBC/PKCS7','compression':'zlib lossless streaming','mac':'chained per-frame HMAC-SHA256 plus final file HMAC','tag':self.filemac.hexdigest(),'records':self.n,'frames':self.frames,'key_id':self.keypath.name,'plaintext_bytes':self.plaintext_bytes,'compressed_bytes':self.compressed_bytes,'ciphertext_bytes':self.ciphertext_bytes,'ciphertext_sha256':self.sha.hexdigest(),'finalized':True}
            self.path.with_suffix(self.path.suffix+'.integrity.json').write_text(json.dumps(result,indent=2));self.integrity=result;return result

def recover_records(path,key,output,integrity=None):
    """PUBLIC verifier. Caller supplies a private key. Stream to private output.
    Missing final integrity/partial frame -> recovered prefix only, never complete.
    """
    digest=hashlib.sha256();filemac=hmac.new(key[32:],digestmod=hashlib.sha256);previous=bytes(32);index=0;z=zlib.decompressobj();partial=False
    def read(f,n):
        b=f.read(n);digest.update(b);filemac.update(b);return b
    with Path(path).open('rb') as f:
        if read(f,8)!=MAGIC:raise BoundaryError('journal magic')
        while True:
            header=read(f,FRAME.size)
            if not header:break
            if len(header)!=FRAME.size:partial=True;break
            seq,size,iv=FRAME.unpack(header)
            if seq!=index+1 or size>33554432 or size<16:raise BoundaryError('frame boundary/index')
            cipher=read(f,size);tag=read(f,32)
            if len(cipher)!=size or len(tag)!=32:partial=True;break
            expected=hmac.new(key[32:],MAGIC+header+previous+cipher,hashlib.sha256).digest()
            if not hmac.compare_digest(tag,expected):raise BoundaryError('frame authentication failed')
            plain=z.decompress(aes_cbc(cipher,key[:32],iv,True));output.write(plain);previous=tag;index=seq
    complete=bool(integrity and not partial and z.eof and hmac.compare_digest(filemac.hexdigest(),integrity['tag']) and digest.hexdigest()==integrity['ciphertext_sha256'] and index==integrity['frames'])
    return {'complete':complete,'authenticated_frames':index,'partial_tail':partial,'ciphertext_sha256':digest.hexdigest()}

def sha256_file(path):
    digest=hashlib.sha256()
    with Path(path).open('rb') as f:
        for chunk in iter(lambda:f.read(1048576),b''):digest.update(chunk)
    return digest.hexdigest()

def validate_native_identity(build):
    path=build.get('build_manifest_path')
    if not path:raise BoundaryError('native build manifest must be explicitly supplied')
    if sha256_file(path)!=build['build_manifest_sha256']:raise BoundaryError('native manifest hash differs')
    manifest=json.loads(Path(path).read_text())
    if manifest.get('kind')!='native' or manifest.get('source_sha')!=build['source_sha'] or manifest.get('source_archive_sha256')!=build['source_archive_sha256'] or manifest.get('runtime_identity')!=build['runtime_identity']:raise BoundaryError('native source/runtime manifest mismatch')
    for k in ['built_utc','python','dependency_lock_sha256','artifact_manifest_sha256','launch_argv_sha256','processes','data_services']:
        if not manifest.get(k):raise BoundaryError('native build/runtime field absent: '+k)
    if manifest.get('image_digest') is not None:raise BoundaryError('native manifest must not claim an image')
    # This binds root-supplied environment provenance; no source or SUT inspection.
    return manifest

def performance_budget_eligible(grant,limit):
    if grant.get('phase')!='performance':return True
    kind=grant.get('run_kind')
    if kind not in ['formal_candidate','diagnostic']:raise BoundaryError('performance must explicitly be formal_candidate or diagnostic')
    if kind=='formal_candidate' and limit<3000000:raise BoundaryError('formal candidate needs >=3000000 request budget; otherwise diagnostic only')
    return kind=='formal_candidate'
