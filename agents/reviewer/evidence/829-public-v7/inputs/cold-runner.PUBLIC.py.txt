"""PUBLIC generic adapter: neutral setup is NOT_PROVEN, never a business failure.
No case/oracle content here. The private business functions and thresholds remain
unchanged. Supervisor admission and private frozen-file checks still apply.
"""
from pathlib import Path
import sys
from control import SetupNotProven

def adapt_case_runner(original,not_proven_type):
    def run_case(self,cid,title,source,fn):
        def with_setup_classification():
            try:return fn()
            except SetupNotProven as error:raise not_proven_type(str(error)) from error
        return original(self,cid,title,source,with_setup_classification)
    return run_case

def main():
    root=Path(__file__).resolve().parents[1];sys.path.insert(0,str(root/'private'))
    import suite
    suite.Suite.run_case=adapt_case_runner(suite.Suite.run_case,suite.NotProven)
    suite.main()
if __name__=='__main__':main()
